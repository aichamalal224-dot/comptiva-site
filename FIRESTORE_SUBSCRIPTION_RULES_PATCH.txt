rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isCreator() {
      return signedIn()
        && request.auth.uid == "ELopCSbMMrSTDEMCzNhchqonGci2";
    }

    // Vérifie la fiche d'abonnement du commerçant côté serveur Firestore.
    function hasActiveSubscription(merchantUid) {
      let sub = get(
        /databases/$(database)/documents/merchantSubscriptions/$(merchantUid)
      );

      return sub.exists()
        && sub.data.keys().hasAll(['status'])
        && (
          (
            sub.data.status == 'trial'
            && sub.data.keys().hasAll(['trialEnd'])
            && sub.data.trialEnd is timestamp
            && request.time < sub.data.trialEnd
          )
          ||
          (
            sub.data.status in ['active', 'paid']
            && sub.data.keys().hasAll(['paidUntil'])
            && sub.data.paidUntil is timestamp
            && request.time < sub.data.paidUntil
          )
        );
    }

    // Données privées : chaque commerçant ne voit que ses propres données.
    // La lecture reste autorisée après expiration pour préserver l'accès aux données.
    // Toute écriture cloud exige un essai ou un abonnement encore valide.
    // Le Créateur n'est exempté que pour ses propres données sous /users/{creatorUid}.
    // Il ne peut pas lire les données commerciales d'un autre commerçant via cette règle.
    match /users/{userId}/{document=**} {
      allow read: if signedIn()
        && request.auth.uid == userId;

      allow write: if signedIn()
        && request.auth.uid == userId
        && (isCreator() || hasActiveSubscription(userId));
    }

    // Annuaire : le Créateur peut consulter et supprimer les fiches.
    // Le commerçant peut créer/actualiser uniquement sa propre fiche de répertoire.
    // Ce répertoire sert au suivi du compte et n'autorise pas l'accès aux données commerciales.
    match /merchantDirectory/{merchantUid} {
      allow read, delete: if isCreator();

      allow create: if signedIn()
        && request.auth.uid == merchantUid
        && request.resource.data.keys().hasOnly([
          'uid', 'email', 'displayName',
          'createdAt', 'updatedAt'
        ])
        && request.resource.data.uid == merchantUid;

      allow update: if signedIn()
        && request.auth.uid == merchantUid
        && request.resource.data.diff(resource.data)
          .affectedKeys().hasOnly([
            'email', 'displayName', 'updatedAt'
          ])
        && request.resource.data.uid == merchantUid;
    }

    // Abonnements : le commerçant lit uniquement sa propre fiche.
    // Il peut créer une seule fiche d'essai de 14 jours.
    // Seul le Créateur peut ensuite modifier ou supprimer une fiche.
    match /merchantSubscriptions/{merchantUid} {
      allow read, write: if isCreator();

      allow get: if signedIn()
        && request.auth.uid == merchantUid;

      allow create: if signedIn()
        && request.auth.uid == merchantUid
        && request.resource.data.keys().hasOnly([
          'uid', 'status', 'trialStart', 'trialEnd',
          'createdAt', 'updatedAt'
        ])
        && request.resource.data.keys().hasAll([
          'uid', 'status', 'trialStart', 'trialEnd',
          'createdAt', 'updatedAt'
        ])
        && request.resource.data.uid == merchantUid
        && request.resource.data.status == 'trial'
        && request.resource.data.trialStart == request.time
        && request.resource.data.createdAt == request.time
        && request.resource.data.updatedAt == request.time
        && request.resource.data.trialEnd is timestamp
        && request.resource.data.trialEnd > request.time
        && request.resource.data.trialEnd
          <= request.time + duration.value(14, 'd');

      allow update, delete: if isCreator();
    }
  }
}

COMPARAISON AVEC LES RÈGLES COMMUNIQUÉES PRÉCÉDEMMENT
- La dernière version que l'utilisateur a collée autorisait déjà chaque commerçant à lire et écrire ses propres documents sous /users/{uid}, sans condition d'abonnement.
- Elle autorisait le commerçant à lire sa fiche merchantSubscriptions, mais réservait sa création, sa modification et sa suppression au Créateur.
- La présente proposition conserve la lecture des propres données après expiration, limite les écritures sous /users/{uid} à un essai/abonnement valide et autorise la création initiale d'un seul document d'essai de 14 jours par le commerçant.
- Avant toute publication, vérifier dans Firebase Console que ces règles communiquées correspondent toujours aux règles réellement actives. L'utilisateur n'a pas confirmé explicitement la publication de cette version.
